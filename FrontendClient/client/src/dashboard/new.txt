import { useMemo } from "react"
import { BRANCHES, type OpEvent, toDateKey } from "'lib/funeral-data"' (see below for file content)

export type CalView = "month" | "week" | "day"

interface CalendarProps {
  view: CalView
  onViewChange: (v: CalView) => void
  cursor: Date // month/week anchor
  onCursorChange: (d: Date) => void
  selected: Date
  onSelect: (d: Date) => void
  events: OpEvent[]
}

const WEEKDAYS = ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"]
const MONTHS = ["January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December"]

function startOfWeek(d: Date) {
  const r = new Date(d)
  const day = (r.getDay() + 6) % 7 // Monday = 0
  r.setDate(r.getDate() - day)
  r.setHours(0, 0, 0, 0)
  return r
}

function sameDay(a: Date, b: Date) {
  return a.getFullYear() === b.getFullYear() && a.getMonth() === b.getMonth() && a.getDate() === b.getDate()
}

export function OperationsCalendar({
  view,
  onViewChange,
  cursor,
  onCursorChange,
  selected,
  onSelect,
  events,
}: CalendarProps) {
  const today = useMemo(() => new Date(), [])

  const byDay = useMemo(() => {
    const map = new Map<string, OpEvent[]>()
    for (const e of events) {
      const arr = map.get(e.date) ?? []
      arr.push(e)
      map.set(e.date, arr)
    }
    return map
  }, [events])

  const monthDays = useMemo(() => {
    const first = new Date(cursor.getFullYear(), cursor.getMonth(), 1)
    const gridStart = startOfWeek(first)
    return Array.from({ length: 42 }, (_, i) => {
      const d = new Date(gridStart)
      d.setDate(gridStart.getDate() + i)
      return d
    })
  }, [cursor])

  const weekDays = useMemo(() => {
    const start = startOfWeek(cursor)
    return Array.from({ length: 7 }, (_, i) => {
      const d = new Date(start)
      d.setDate(start.getDate() + i)
      return d
    })
  }, [cursor])

  function shift(dir: number) {
    const d = new Date(cursor)
    if (view === "month") d.setMonth(d.getMonth() + dir)
    else if (view === "week") d.setDate(d.getDate() + dir * 7)
    else d.setDate(d.getDate() + dir)
    onCursorChange(d)
    if (view === "day") onSelect(d)
  }

  const headerLabel =
    view === "month"
      ? `${MONTHS[cursor.getMonth()]} ${cursor.getFullYear()}`
      : view === "week"
        ? `Week of ${weekDays[0].getDate()} ${MONTHS[weekDays[0].getMonth()].slice(0, 3)} – ${weekDays[6].getDate()} ${MONTHS[weekDays[6].getMonth()].slice(0, 3)}`
        : `${selected.toLocaleDateString("en-GB", { weekday: "long", day: "numeric", month: "long" })}`

  function branchDots(d: Date) {
    const list = byDay.get(toDateKey(d)) ?? []
    const set = new Set(list.map((e) => e.branch))
    return Array.from(set)
  }

  return (
    <div className="flex h-full flex-col rounded-xl border border-border bg-card">
      {/* Toolbar */}
      <div className="flex flex-wrap items-center justify-between gap-3 border-b border-border p-4">
        <div className="flex items-center gap-2">
          <Button variant="outline" size="icon" className="h-8 w-8 bg-transparent" onClick={() => shift(-1)} aria-label="Previous">
            <ChevronLeft className="h-4 w-4" />
          </Button>
          <Button variant="outline" size="icon" className="h-8 w-8 bg-transparent" onClick={() => shift(1)} aria-label="Next">
            <ChevronRight className="h-4 w-4" />
          </Button>
          <h2 className="ml-1 text-sm font-semibold text-foreground">{headerLabel}</h2>
          <Button
            variant="ghost"
            size="sm"
            className="h-7 text-xs text-muted-foreground"
            onClick={() => {
              onCursorChange(new Date())
              onSelect(new Date())
            }}
          >
            Today
          </Button>
        </div>

        <div className="flex items-center rounded-lg border border-border p-0.5">
          {(["month", "week", "day"] as CalView[]).map((v) => (
            <button
              key={v}
              onClick={() => onViewChange(v)}
              className={cn(
                "rounded-md px-3 py-1 text-xs font-medium capitalize transition-colors",
                view === v ? "bg-foreground text-background" : "text-muted-foreground hover:text-foreground",
              )}
            >
              {v}
            </button>
          ))}
        </div>
      </div>

      {/* Month view */}
      {view === "month" && (
        <div className="flex flex-1 flex-col p-3">
          <div className="grid grid-cols-7">
            {WEEKDAYS.map((d) => (
              <div key={d} className="pb-2 text-center text-[11px] font-medium uppercase tracking-wide text-muted-foreground">
                {d}
              </div>
            ))}
          </div>
          <div className="grid flex-1 grid-cols-7 gap-1">
            {monthDays.map((d, i) => {
              const inMonth = d.getMonth() === cursor.getMonth()
              const isSel = sameDay(d, selected)
              const isToday = sameDay(d, today)
              const dots = branchDots(d)
              const total = (byDay.get(toDateKey(d)) ?? []).length
              return (
                <button
                  key={i}
                  onClick={() => onSelect(d)}
                  className={cn(
                    "flex min-h-[64px] flex-col items-start rounded-lg border p-1.5 text-left transition-colors",
                    isSel ? "border-foreground bg-foreground/[0.03]" : "border-transparent hover:border-border hover:bg-muted/50",
                    !inMonth && "opacity-40",
                  )}
                >
                  <span
                    className={cn(
                      "flex h-6 w-6 items-center justify-center rounded-full text-xs tabular-nums",
                      isToday ? "bg-foreground font-semibold text-background" : "text-foreground",
                    )}
                  >
                    {d.getDate()}
                  </span>
                  <div className="mt-auto flex w-full items-center justify-between pt-1">
                    <div className="flex gap-0.5">
                      {dots.map((b) => (
                        <span key={b} className="h-1.5 w-1.5 rounded-full" style={{ backgroundColor: BRANCHES[b].color }} />
                      ))}
                    </div>
                    {total > 0 && <span className="text-[10px] font-medium tabular-nums text-muted-foreground">{total}</span>}
                  </div>
                </button>
              )
            })}
          </div>
        </div>
      )}

      {/* Week view */}
      {view === "week" && (
        <div className="grid flex-1 grid-cols-7 gap-1 p-3">
          {weekDays.map((d, i) => {
            const isSel = sameDay(d, selected)
            const isToday = sameDay(d, today)
            const list = (byDay.get(toDateKey(d)) ?? []).slice().sort((a, b) => a.time.localeCompare(b.time))
            return (
              <button
                key={i}
                onClick={() => onSelect(d)}
                className={cn(
                  "flex flex-col rounded-lg border p-2 text-left transition-colors",
                  isSel ? "border-foreground bg-foreground/[0.03]" : "border-border hover:bg-muted/50",
                )}
              >
                <div className="mb-2 flex items-center justify-between">
                  <span className="text-[11px] font-medium uppercase text-muted-foreground">{WEEKDAYS[i]}</span>
                  <span
                    className={cn(
                      "flex h-6 w-6 items-center justify-center rounded-full text-xs tabular-nums",
                      isToday ? "bg-foreground font-semibold text-background" : "text-foreground",
                    )}
                  >
                    {d.getDate()}
                  </span>
                </div>
                <div className="flex flex-col gap-1">
                  {list.slice(0, 6).map((e) => (
                    <div
                      key={e.id}
                      className="truncate rounded px-1.5 py-1 text-[10px] font-medium"
                      style={{ backgroundColor: BRANCHES[e.branch].color + "1a", color: BRANCHES[e.branch].color }}
                    >
                      <span className="tabular-nums">{e.time}</span> {e.title}
                    </div>
                  ))}
                  {list.length > 6 && <span className="px-1 text-[10px] text-muted-foreground">+{list.length - 6} more</span>}
                </div>
              </button>
            )
          })}
        </div>
      )}

      {/* Day view */}
      {view === "day" && <DayTimeline events={(byDay.get(toDateKey(selected)) ?? [])} />}
    </div>
  )
}

function DayTimeline({ events }: { events: OpEvent[] }) {
  const hours = Array.from({ length: 14 }, (_, i) => i + 7) // 07:00 - 20:00
  return (
    <div className="flex-1 overflow-auto p-3">
      <div className="flex flex-col">
        {hours.map((h) => {
          const slot = events
            .filter((e) => Number.parseInt(e.time.split(":")[0], 10) === h)
            .sort((a, b) => a.time.localeCompare(b.time))
          return (
            <div key={h} className="flex gap-3 border-b border-border py-2 last:border-0">
              <span className="w-12 shrink-0 pt-0.5 font-mono text-[11px] tabular-nums text-muted-foreground">
                {h.toString().padStart(2, "0")}:00
              </span>
              <div className="flex flex-1 flex-wrap gap-1.5">
                {slot.length === 0 ? (
                  <span className="text-[11px] text-muted-foreground/40">—</span>
                ) : (
                  slot.map((e) => (
                    <div
                      key={e.id}
                      className="rounded-md border px-2 py-1 text-[11px]"
                      style={{
                        backgroundColor: BRANCHES[e.branch].color + "12",
                        borderColor: BRANCHES[e.branch].color + "40",
                        color: BRANCHES[e.branch].color,
                      }}
                    >
                      <span className="tabular-nums">{e.time}</span> · {e.title} · {e.deceasedName}
                    </div>
                  ))
                )}
              </div>
            </div>
          )
        })}
      </div>
    </div>
  )
}

              const over = u.value >= 90
              return (
                <li key={u.branch}>
                  <div className="mb-1 flex items-center justify-between text-xs">
                    <span className="font-medium text-foreground">{u.name}</span>
                    <span className="tabular-nums" style={{ color: over ? "#dc2626" : branch.color }}>
                      {u.value}%
                    </span>
                  </div>
                  <div className="h-2 w-full overflow-hidden rounded-full bg-muted">
                    <div
                      className="h-full rounded-full transition-[width] duration-700"
                      style={{ width: `${u.value}%`, backgroundColor: over ? "#dc2626" : branch.color }}
                    />
                  </div>
                </li>
              )
            })}
          </ul>

          <div className="mt-5 rounded-lg border border-border p-3">
            <p className="text-xs font-medium text-muted-foreground">Fleet fuel average</p>
            <p className="mt-1 text-2xl font-semibold tabular-nums text-foreground">{fleetFuelAvg()}%</p>
            <div className="mt-2 flex flex-col gap-1.5">
              {(["nairobi", "thika", "machakos"] as const).map((b) => (
                <div key={b} className="flex items-center justify-between text-[11px]">
                  <span className="flex items-center gap-1.5 text-muted-foreground">
                    <span className="h-2 w-2 rounded-full" style={{ backgroundColor: BRANCHES[b].color }} />
                    {BRANCHES[b].name}
                  </span>
                  <span className="tabular-nums text-foreground">{fleetFuelAvg(b)}%</span>
                </div>
              ))}
            </div>
          </div>
        </div>
      </div>
    </section>
  )
}
